n = int(input())#4
ranges = []
for i in range(n):
    inp = input().split(" ")
    start = int(inp[0])
    end = int(inp[1])
    ranges.append([start,end])
#[[1,3],[2,6],[8,10],[15,18]]
res_ranges =[]
start = ranges[0][0]
end = ranges[0][1]
for i in range(1,len(ranges)):
    if (ranges[i][0]>=start and ranges[i][0]<=end) and (ranges[i][1]>=end):
        end = ranges[i][1]
        res_ranges.append([start,end])
    else:
        start = ranges[i][0]
        end = ranges[i][1]
        res_ranges.append([start,end])
for i in range(len(res_ranges)):
    print(res_ranges[i][0],res_ranges[i][1])
